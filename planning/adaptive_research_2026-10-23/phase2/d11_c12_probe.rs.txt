const TOOL_RADIUS: f64 = 3.0;
const STEPOVER: f64 = 2.0;
const TOLERANCE: f64 = 0.1;
const DEPTH_PER_PASS: f64 = 3.0;
const STOCK_TO_LEAVE: f64 = 0.5;
const TOP_Z: f64 = 10.0;
const FLOOR_Z: f64 = 4.0;
const MESH_HALF: f64 = 40.0;
const MESH_STEP: f64 = 0.5;
const POCKET_HALF_X: f64 = 34.0;
const POCKET_HALF_Y: f64 = 24.0;
const ISLAND_RADIUS: f64 = 5.0;
const SIM_CELL: f64 = 0.5;
const PERIMETER_INSET_MARGIN_MM: f64 = 0.25;
const EXACT_PITCH_MM: f64 = 0.05;

fn pocket_with_island() -> TriangleMesh {
    height_field(MESH_HALF, MESH_STEP, |x, y| {
        if x.hypot(y) <= ISLAND_RADIUS {
            TOP_Z
        } else if x.abs() <= POCKET_HALF_X && y.abs() <= POCKET_HALF_Y {
            FLOOR_Z
        } else {
            TOP_Z
        }
    })
}

fn planner_cell() -> f64 {
    (TOOL_RADIUS / 6.0).max(TOLERANCE)
}

fn load_bound() -> f64 {
    let ceiling =
        radial_woc_fraction_from_leading_arc(pass_engagement_limit(STEPOVER, TOOL_RADIUS));
    ceiling + (planner_cell() + SIM_CELL + TOLERANCE) / (2.0 * TOOL_RADIUS)
}

fn wall_distance(x: f64, y: f64) -> f64 {
    let sides = (POCKET_HALF_X - x.abs()).min(POCKET_HALF_Y - y.abs());
    let island = x.hypot(y) - ISLAND_RADIUS;
    sides.min(island)
}

fn in_perimeter_band(x: f64, y: f64) -> bool {
    let d = wall_distance(x, y);
    let centre = 2.0 * TOOL_RADIUS + PERIMETER_INSET_MARGIN_MM;
    let quantum = 1.5 * planner_cell();
    d >= centre - quantum && d <= centre + SQRT_2 * MESH_STEP + quantum
}

fn params(strategy: ClearingStrategy3d) -> Adaptive3dParams {
    Adaptive3dParams {
        entry_clearance_mm: 0.5,
        ramp_feed_rate: None,
        geometry: Adaptive3dGeometry {
            tool_radius: TOOL_RADIUS,
            envelope_radius: TOOL_RADIUS,
            stepover: STEPOVER,
            tolerance: TOLERANCE,
            segment_merge_tolerance: None,
            min_cutting_radius: 0.0,
            boundary: None,
            world_stock_xy_bbox: None,
            centre_boundary: Vec::new(),
        },
        depth: Adaptive3dDepth {
            depth_per_pass: DEPTH_PER_PASS,
            stock_to_leave: STOCK_TO_LEAVE,
            stock_top_z: TOP_Z,
            z_floor: None,
            detect_flat_areas: false,
        },
        linking: Adaptive3dLinking {
            region_ordering: RegionOrdering::Global,
            min_region_cut_length_mm: 0.0,
            max_stay_down_distance_mm: None,
            stay_down_clearance_mm: 0.5,
        },
        feed_rate: 1000.0,
        plunge_rate: 500.0,
        safe_z: 15.0,
        entry_style: EntryStyle3d::Plunge,
        initial_stock: None,
        clearing_strategy: strategy,
        trochoid_cap_mult: 1.6,
        engagement_measure: rs_cam_core::adaptive::EngagementMeasure::DiskArea,
        z_blend: false,
    }
}

fn is_entry(intent: MoveIntent) -> bool {
    matches!(
        intent,
        MoveIntent::EntryPlunge | MoveIntent::EntryHelix | MoveIntent::EntryRamp
    )
}

/// The exact swept width of `from -> to` on move `m` (copied from
/// `a_clearing_cut_holds_the_pass_load_g_adaptpassload::exact_width_at`,
/// lines only; arcs read as chords: the arm emits `Linear` moves here).
fn exact_width_at(tp: &Toolpath, m: usize, from: (f64, f64), to: (f64, f64)) -> f64 {
    let r = TOOL_RADIUS;
    let pitch = EXACT_PITCH_MM;
    let z = tp.moves[m].target.z;
    let half = 2.0 * r;
    let n = (2.0 * half / pitch) as usize + 1;
    let (x0, y0) = (to.0 - half, to.1 - half);
    let mut cut = vec![false; n * n];
    let stamp = |cut: &mut Vec<bool>, px: f64, py: f64| {
        if (px - to.0).abs() > half + r || (py - to.1).abs() > half + r {
            return;
        }
        let i0 = ((px - r - x0) / pitch).floor().max(0.0) as usize;
        let i1 = (((px + r - x0) / pitch).ceil().max(0.0) as usize).min(n - 1);
        let j0 = ((py - r - y0) / pitch).floor().max(0.0) as usize;
        let j1 = (((py + r - y0) / pitch).ceil().max(0.0) as usize).min(n - 1);
        for j in j0..=j1 {
            for i in i0..=i1 {
                let (x, y) = (x0 + i as f64 * pitch, y0 + j as f64 * pitch);
                if (x - px).hypot(y - py) <= r {
                    cut[j * n + i] = true;
                }
            }
        }
    };
    let pts = |a: rs_cam_core::geo::P3, b: rs_cam_core::geo::P3| -> Vec<(f64, f64)> {
        let len = (b.x - a.x).hypot(b.y - a.y) * 2.0;
        let k = ((len / pitch).ceil() as usize).max(1);
        (0..=k)
            .map(|s| {
                let t = s as f64 / k as f64;
                (a.x + t * (b.x - a.x), a.y + t * (b.y - a.y))
            })
            .collect()
    };
    let mut prev = tp.moves[0].target;
    for mv in tp.moves.iter().take(m) {
        if mv.target.z <= z + 0.01 && prev.z <= z + 0.01 {
            for (px, py) in pts(prev, mv.target) {
                stamp(&mut cut, px, py);
            }
        } else if mv.target.z <= z + 0.01 {
            stamp(&mut cut, mv.target.x, mv.target.y);
        }
        prev = mv.target;
    }
    for (px, py) in pts(prev, tp.moves[m].target) {
        stamp(&mut cut, px, py);
        if (px - from.0).hypot(py - from.1) < pitch {
            break;
        }
    }
    let (dx, dy) = (to.0 - from.0, to.1 - from.1);
    let l = dx.hypot(dy).max(1e-9);
    let (sx, sy) = (-dy / l, dx / l);
    let mid = (0.5 * (from.0 + to.0), 0.5 * (from.1 + to.1));
    let (mut lo, mut hi) = (f64::INFINITY, f64::NEG_INFINITY);
    for j in 0..n {
        for i in 0..n {
            let (x, y) = (x0 + i as f64 * pitch, y0 + j as f64 * pitch);
            if !cut[j * n + i] && (x - mid.0).hypot(y - mid.1) <= r {
                let u = (x - mid.0) * sx + (y - mid.1) * sy;
                lo = lo.min(u);
                hi = hi.max(u);
            }
        }
    }
    if hi > lo { (hi - lo) / (2.0 * r) } else { 0.0 }
}

fn kind(mt: &MoveType) -> &'static str {
    match mt {
        MoveType::Rapid => "rapid",
        MoveType::Linear { .. } => "lin",
        MoveType::ArcCW { .. } | MoveType::ArcCCW { .. } => "arc",
    }
}

/// The length of the run of collinear `ClearingCut` moves of one planner
/// cell that holds move `m` (the signature of `links_as_cuts`: a link
/// subdivided at the planner cell), as (first, last, chain mm).
fn cell_chain(tp: &Toolpath, m: usize) -> Option<(usize, usize, f64)> {
    let cell = planner_cell();
    let seg = |k: usize| -> Option<(f64, f64, f64)> {
        if k == 0 || k >= tp.moves.len() {
            return None;
        }
        let (a, b) = (tp.moves[k - 1].target, tp.moves[k].target);
        let (dx, dy) = (b.x - a.x, b.y - a.y);
        let l = dx.hypot(dy);
        (l > 1e-6).then_some((dx / l, dy / l, l))
    };
    let (ux, uy, l) = seg(m)?;
    if l > cell + 1e-6 {
        return None;
    }
    let same = |k: usize| {
        seg(k).is_some_and(|(vx, vy, lk)| {
            lk <= cell + 1e-6 && (ux * vx + uy * vy) > 1.0 - 1e-9 && tp.moves[k].intent == tp.moves[m].intent
        })
    };
    let mut first = m;
    while first > 1 && same(first - 1) {
        first -= 1;
    }
    let mut last = m;
    while last + 1 < tp.moves.len() && same(last + 1) {
        last += 1;
    }
    let (a, b) = (tp.moves[first - 1].target, tp.moves[last].target);
    let chain = (b.x - a.x).hypot(b.y - a.y);
    // A chain of two or more cell steps; a single short step is not a link.
    (last > first).then_some((first, last, chain))
}

#[test]
#[ignore = "instrument: D11 scratch probe"]
fn d11_classify_c12_over_samples() {
    let mesh = pocket_with_island();
    let index = SpatialIndex::build(&mesh, 10.0);
    let bound = load_bound();
    let cutter = FlatEndmill::new(2.0 * TOOL_RADIUS, 25.0);
    for (label, strategy) in [
        ("agent", ClearingStrategy3d::AgentSearch),
        ("geodesic_curv", ClearingStrategy3d::Geodesic(GeodesicOptions::C5_CURV)),
        ("geodesic_curv_split", ClearingStrategy3d::Geodesic(GeodesicOptions::C11_SPLIT)),
    ] {
        let tp = adaptive_3d_toolpath_output_with_cancel(
            &mesh,
            &index,
            &cutter,
            &params(strategy),
            &|| false,
            None,
        )
        .expect("plan")
        .toolpath;
        let b = &mesh.bbox;
        let mut stock = TriDexelStock::from_stock(
            b.min.x - TOOL_RADIUS,
            b.min.y - TOOL_RADIUS,
            b.max.x + TOOL_RADIUS,
            b.max.y + TOOL_RADIUS,
            b.min.z,
            TOP_Z,
            SIM_CELL,
        );
        let samples = stock
            .simulate_toolpath_with_metrics_with_cancel(
                &tp,
                &cutter,
                StockCutDirection::FromTop,
                ToolpathId::from_raw(0),
                12_000,
                2,
                3000.0,
                SIM_CELL,
                None,
                &[],
                &[],
                false,
                &|| false,
            )
            .expect("sim");
        // The judge of the C12 test, then the over samples with their context.
        let mut hole: Option<(f64, f64)> = None;
        let mut last_entry: Option<usize> = None;
        let mut over: Vec<(usize, [f64; 3], f64, Option<usize>)> = Vec::new();
        for s in &samples {
            let Some(mv) = tp.moves.get(s.move_index) else {
                continue;
            };
            if is_entry(mv.intent) {
                last_entry = Some(s.move_index);
            }
            if mv.intent == MoveIntent::EntryPlunge {
                hole = Some((mv.target.x, mv.target.y));
            }
            if !s.is_cutting {
                continue;
            }
            let [x, y, _] = s.position;
            let radial = s.engagement.radial_woc_fraction;
            if is_entry(mv.intent) {
                continue;
            }
            if let Some((hx, hy)) = hole {
                if (x - hx).hypot(y - hy) <= TOOL_RADIUS {
                    continue;
                }
                hole = None;
            }
            if in_perimeter_band(x, y) {
                continue;
            }
            if radial > bound {
                over.push((s.move_index, s.position, radial, last_entry));
            }
        }
        eprintln!("=== {label}: {} moves, {} over samples (bound {bound:.4})", tp.moves.len(), over.len());
        // Per over move: the exact peak of the move on the emitted path.
        let mut exact_of: std::collections::BTreeMap<usize, f64> = Default::default();
        for &(m, ..) in &over {
            exact_of.entry(m).or_insert_with(|| {
                let mut last = (tp.moves[m - 1].target.x, tp.moves[m - 1].target.y);
                let mut peak = 0.0_f64;
                for s in samples.iter().filter(|s| s.move_index == m) {
                    let at = (s.position[0], s.position[1]);
                    if (at.0 - last.0).hypot(at.1 - last.1) > 1e-6 {
                        peak = peak.max(exact_width_at(&tp, m, last, at));
                    }
                    last = at;
                }
                peak
            });
        }
        let mut rows: Vec<String> = Vec::new();
        for &(m, p, radial, le) in &over {
            let mv = &tp.moves[m];
            let a = tp.moves[m - 1].target;
            let len = (mv.target.x - a.x).hypot(mv.target.y - a.y);
            let dz = mv.target.z - a.z;
            let chain = cell_chain(&tp, m);
            let since_entry = le.map(|e| m - e);
            let entry_xy = le.map(|e| {
                let t = tp.moves[e].target;
                (p[0] - t.x).hypot(p[1] - t.y)
            });
            let ctx: Vec<String> = (m.saturating_sub(3)..(m + 3).min(tp.moves.len()))
                .map(|k| {
                    let t = tp.moves[k].target;
                    format!("{k}:{:?}/{}@({:.2},{:.2},{:.2})", tp.moves[k].intent, kind(&tp.moves[k].move_type), t.x, t.y, t.z)
                })
                .collect();
            rows.push(format!(
                "ROW {label} move {m} {:?}/{} sample ({:.2},{:.2},{:.2}) wall {:.2} radial {radial:.3} exact_move_peak {:.3} | move from ({:.2},{:.2},{:.2}) len {len:.3} dz {dz:+.3} | cell_chain {} | since_entry {} entry_xy {} | ctx {}",
                mv.intent,
                kind(&mv.move_type),
                p[0],
                p[1],
                p[2],
                wall_distance(p[0], p[1]),
                exact_of[&m],
                a.x,
                a.y,
                a.z,
                chain.map_or("none".to_string(), |(f, l, c)| format!("{f}..{l} {c:.2}mm")),
                since_entry.map_or("none".to_string(), |v| v.to_string()),
                entry_xy.map_or("none".to_string(), |v| format!("{v:.2}")),
                ctx.join(" "),
            ));
        }
        for r in rows {
            eprintln!("{r}");
        }
    }
}
